import asyncio
import gc
import json
import sys
import time
import tracemalloc
from pathlib import Path
from types import SimpleNamespace

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT / 'src'))
from animetta.orchestration.graph.conversation_session import ConversationSessionState
from animetta.orchestration.server.handlers.chat_handlers import ChatHandlers
from animetta.services.bilibili.models import DanmakuMessage
from animetta.services.bilibili.reply_admission import ReplyPriority
from animetta.services.bilibili.reply_queue import ReplyCandidate, ReplyMetrics, ReplyWorker

class ProbeQueue:
    qsize = 0
    def __init__(self, count):
        self.count = count
        self.index = 0
    async def get(self):
        if self.index == self.count:
            return None
        self.index += 1
        return ReplyCandidate(message=DanmakuMessage(text='sample', timestamp=0), priority=ReplyPriority.ORDINARY, generation_id=1, admitted_at=0, reply_id=str(self.index))

async def noop(*args, **kwargs):
    return None

async def measure(count):
    gc.collect()
    tracemalloc.start()
    started = time.perf_counter()
    state = ConversationSessionState()
    metrics = ReplyMetrics()
    chat = ChatHandlers(None, None, None)
    chat._process_text_command = noop
    for index in range(count):
        state.commit(task_id=f'task-{index}', user_text='user', final_response='reply')
        await chat.on_text_command('sid', SimpleNamespace(conversation_id=f'conversation-{index}'))
    worker = ReplyWorker(queue=ProbeQueue(count), processor=noop, metrics=metrics, generation_id=1, max_message_age_seconds=None, clock=lambda:1.0)
    await worker.run()
    gc.collect()
    retained, peak = tracemalloc.get_traced_memory()
    result = {'turns':count,'conversation_window':len(state.completed_turns),'dedup_ids':len(state._committed_tasks),'queue_wait_samples':len(metrics.queue_wait_seconds),'reply_latency_samples':len(metrics.reply_latency_seconds),'chat_locks':len(chat._conversation_locks),'success_count':metrics.reply_success,'retained_bytes':retained,'peak_bytes':peak,'elapsed_seconds':time.perf_counter()-started}
    tracemalloc.stop()
    return result

async def main():
    samples = [await measure(count) for count in (1000,10000,30000) for _ in range(3)]
    result={'kind':'isolated_retention_probe','condition':'same process, fresh state per sample, real commit/chat-lock/reply-worker code, no external services; tracemalloc excludes imported modules','samples':samples}
    output = ROOT/'.zcode/memory-retention-20261014'/sys.argv[1]
    output.write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(result))

asyncio.run(main())
