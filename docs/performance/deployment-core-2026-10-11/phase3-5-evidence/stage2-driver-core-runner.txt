import hashlib
import json
import os
import re
import socket
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path
from urllib.request import Request, urlopen

import psutil
from dotenv import dotenv_values

ROOT = Path(r'C:\Users\30262\Project\Anima')
REPORT = ROOT / 'docs/performance/deployment-core-2026-10-11'
BASE = json.loads((REPORT / 'runtime-baseline.json').read_text(encoding='utf-8'))
CANDIDATE = json.loads((REPORT / 'runtime-core.json').read_bytes())['prepared_core_r3_input']
SOURCE = Path(CANDIDATE['source_root'])
SCRATCH = SOURCE.parent
MEASURE = SCRATCH / 'measurements'
MEASURE.mkdir(exist_ok=True)
ENV = os.environ.copy()
ENV.update({k: v for k, v in dotenv_values(ROOT / '.env').items() if v is not None})
ENV.update(CANDIDATE['future_execution_environment_overrides'])
SECRETS = [v for k, v in ENV.items() if v and re.search(r'PASSWORD|TOKEN|SECRET|API_KEY|COOKIE', k, re.I)]


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def write(path, obj):
    raw = json.dumps(obj, ensure_ascii=False, indent=2) + '\n'
    for value in SECRETS:
        if len(value) >= 8:
            raw = raw.replace(value, '<redacted>')
    path.write_text(raw, encoding='utf-8')


def command(argv):
    result = subprocess.run(argv, cwd=SOURCE, env=ENV, capture_output=True, text=True, encoding='utf-8', errors='replace', timeout=30)
    assert result.returncode == 0, 'Read-only command failed: ' + str(argv[:3])
    return result.stdout


def http(url, key='ANIMETTA_ACCESS_TOKEN'):
    with urlopen(Request(url, headers={'Authorization': 'Bearer ' + ENV[key]}), timeout=20) as response:
        raw = response.read()
        return {'status': response.status, 'bytes': len(raw), 'sha256': sha(raw), 'body': json.loads(raw) if raw.startswith(b'{') else None,
                'assets': re.findall(r'(?:src|href)=[\"\']([^\"\']+\.(?:js|css)[^\"\']*)', raw.decode('utf-8', 'replace'))}


def process(p):
    return {'pid': p.pid, 'creation_time': p.create_time(), 'command_line': p.cmdline(), 'rss_bytes': p.memory_info().rss}


def ownership():
    processes = []
    for expected in CANDIDATE['qwen_preserved']:
        p = psutil.Process(expected['pid'])
        assert abs(p.create_time() - expected['creation_time']) < 0.01
        assert p.cmdline() == expected['command_line']
        processes.append(process(p))
    competing = []
    for p in psutil.process_iter(['name', 'cmdline']):
        try:
            if p.pid == os.getpid():
                continue
            a = p.info['cmdline'] or []
            n = (p.info['name'] or '').lower()
            lifecycle = any(Path(x).name in {'runtime_lifecycle.py', 'benchmark_startup.py'} for x in a)
            quality = n.startswith('python') and '-m' in a and any(x in a for x in ['tooling.quality', 'pytest', 'mypy', 'pip'])
            install = n in {'uv.exe', 'pnpm.exe'} and any(x in a for x in ['sync', 'install', 'test'])
            frontend = n == 'node.exe' and any(('vitest' in x or 'vue-tsc' in x) for x in a)
            training = n.startswith('python') and any(re.search(r'(?:^|[\\/])(train|training)[^\\/]*\.py$', x, re.I) for x in a)
            if lifecycle or quality or install or frontend or training:
                competing.append(process(p))
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    assert not competing, 'Competing process detected'
    with socket.socket() as s:
        s.settimeout(2)
        assert s.connect_ex(('127.0.0.1', 8769)) != 0
    identity = {ep: http('http://127.0.0.1:8767' + ep, 'QWEN_TTS_API_KEY')['body'] for ep in ['/ready', '/v1/identity']}
    assert identity == BASE['qwen_ready_identity']
    return {'qwen_processes': processes, 'qwen_identity': identity, 'rvc_port_8769_unavailable': True, 'competing_processes': competing}


def preflight():
    assert not (REPORT / 'runtime-core-r3.json').exists(), 'Read existing r3 run rather than repeat preflight/reset record'
    assert sys.version_info >= (3, 13)
    manifest = json.loads(Path(CANDIDATE['manifest_path']).read_bytes())
    assert sha(Path(CANDIDATE['archive']).read_bytes()) == CANDIDATE['archive_sha256']
    assert sha(Path(CANDIDATE['manifest_path']).read_bytes()) == CANDIDATE['manifest_sha256']
    assert all(sha((SOURCE / p).read_bytes()) == h for p, h in manifest['files'].items())
    assert all(sha((ROOT / p).read_bytes()) == h for p, h in CANDIDATE['overlay_files'].items())
    assert sha(Path(CANDIDATE['task_paths_source']).read_bytes()) == CANDIDATE['task_paths_sha256']
    assert sha((ROOT / '.env').read_bytes()) == CANDIDATE['original_env_sha256']
    assert all(ENV.get(k) for k in ['QWEN_TTS_API_KEY', 'ANIMETTA_REDIS_PASSWORD', 'ANIMETTA_ACCESS_TOKEN'])
    gate_path = ROOT / 'artifacts/test-impact/108051f2a0bda55067912968074a51a086decfe54d9877795c4651af7fc5096c/results/summary.json'
    gate = json.loads(gate_path.read_bytes())
    assert gate['status'] == 'passed'
    assert all(not gate[k] for k in ['failed_groups', 'blocked_groups', 'missing_groups', 'cancelled_groups'])
    value = ownership()
    value.update({'at': datetime.now(UTC).isoformat(), 'frozen_files_verified': 2097, 'overlay_files_verified': 42,
                  'task_paths_verified': 75, 'env_hash_verified': True, 'gate': gate, 'gate_summary_sha256': sha(gate_path.read_bytes()),
                  'gpu': command(['nvidia-smi', '--query-gpu=name,memory.total,memory.used,memory.free', '--format=csv,noheader,nounits']).strip(),
                  'gpu_compute_apps': command(['nvidia-smi', '--query-compute-apps=pid,process_name,used_memory', '--format=csv,noheader,nounits']).strip(),
                  'existing_http': {ep: http('http://localhost' + ep) for ep in ['/health', '/ready', '/dashboard', '/live.html']},
                  'existing_app_image': command(['docker', 'inspect', '--format', '{{.Image}}', 'anima-animetta-1']).strip(),
                  'existing_app_state': json.loads(command(['docker', 'inspect', '--format', '{{json .State}}', 'anima-animetta-1'])),
                  'quality_install_overlap': False, 'secret_values_persisted': False})
    value['gpu_process_commands'] = [process(psutil.Process(int(line.split(',')[0]))) for line in value['gpu_compute_apps'].splitlines() if line]
    command(['docker', 'compose', 'config', '--quiet'])
    value['compose_config_quiet_exit_code'] = 0
    write(MEASURE / 'core-preflight.json', value)
    core = {'schema_version': 1, 'status': 'preflight_passed', 'started_at': value['at'], 'frozen_input': CANDIDATE,
            'authorization': {'received': True, 'gate_status': 'passed', 'gate_plan_hash': gate['plan_hash'], 'gate_evidence': str(gate_path)},
            'preflight': value, 'sample_definitions': BASE['sample_definitions'], 'runs': [],
            'samples': {'application_cold': [], 'application_restart_qwen_reused': []}, 'resource_profiles': [],
            'planned_acceptance': BASE['planned_core_acceptance'], 'comparability': {'quality_install_overlap': False, 'baseline_quality_cpu_overlap': True, 'hardware_disk_cache_cold': False, 'build_cache_retained': True, 'clean_start_speedup_claim_supported': False},
            'resume': {'scratch_root': str(SCRATCH), 'completed_runs': [], 'active_run_id': None, 'next_action': 'c-dc2-down01 then c-dc2-app01'}}
    write(REPORT / 'runtime-core-r3.json', core)
    print(json.dumps({'preflight': 'passed', 'qwen_pids': [x['pid'] for x in value['qwen_processes']], 'gpu': value['gpu'], 'existing_image': value['existing_app_image']}))


def lifecycle(operation, run_id):
    ownership()
    core = json.loads((REPORT / 'runtime-core-r3.json').read_bytes())
    assert run_id not in core['resume']['completed_runs'], 'Never repeat completed run'
    argv = ['py', '-3.13', str(SOURCE / 'scripts/benchmark_startup.py'), 'worker', '--root', str(SOURCE), '--operation', operation, '--run-id', run_id, '--phase', 'after', '--artifacts-root', str(ROOT / 'artifacts/iteration-plans'), '--output', str(MEASURE / (run_id + '.json'))]
    now = datetime.now(UTC).isoformat()
    core['status'] = 'running'
    core['resume'].update({'active_run_id': run_id, 'next_action': 'Read existing run receipt/result, never start another run before terminal.'})
    core['runs'].append({'run_id': run_id, 'operation': operation, 'command': argv, 'started_at': now, 'quality_install_overlap': False, 'output': str(MEASURE / (run_id + '.json')), 'outer_log': str(MEASURE / (run_id + '.log'))})
    write(REPORT / 'runtime-core-r3.json', core)
    with (REPORT / 'runtime-log.md').open('a', encoding='utf-8') as f:
        f.write(f'\n- Core run `{run_id}` operation `{operation}` started {now}; canonical lifecycle worker on frozen r3, Qwen ownership preserved, 8769 unavailable, no quality/install competitors.\n')
    with (MEASURE / (run_id + '.log')).open('w', encoding='utf-8') as log:
        result = subprocess.run(argv, cwd=SOURCE, env=ENV, stdout=log, stderr=subprocess.STDOUT)
    core = json.loads((REPORT / 'runtime-core-r3.json').read_bytes())
    entry = core['runs'][-1]
    entry.update({'finished_at': datetime.now(UTC).isoformat(), 'exit_code': result.returncode})
    if (MEASURE / (run_id + '.json')).is_file():
        entry['raw_metrics'] = json.loads((MEASURE / (run_id + '.json')).read_bytes())
    entry['status'] = 'passed' if result.returncode == 0 else 'failed'
    core['status'] = entry['status']
    core['resume']['active_run_id'] = None
    if result.returncode == 0:
        core['resume']['completed_runs'].append(run_id)
    core['resume']['next_action'] = 'Collect this completed run evidence before next sample.' if result.returncode == 0 else 'Terminal failure: diagnose original run, no automatic retry.'
    write(REPORT / 'runtime-core-r3.json', core)
    print(json.dumps({'run_id': run_id, 'operation': operation, 'exit_code': result.returncode, 'status': entry['status']}))
    return result.returncode


if __name__ == '__main__':
    if sys.argv[1] == 'preflight':
        preflight()
    elif sys.argv[1] == 'lifecycle':
        sys.exit(lifecycle(sys.argv[2], sys.argv[3]))
