"""Resume only acceptance after original canonical exit0; no lifecycle call here."""
import ast
import importlib.util
import json
from datetime import UTC, datetime
from pathlib import Path

spec=importlib.util.spec_from_file_location('distribution',Path(__file__).with_name('distribution-driver.py'))
d=importlib.util.module_from_spec(spec)
spec.loader.exec_module(d)


def receipts():
    history=d.OUT/'collector-HOLD-original.json'
    original=history.read_bytes() if history.exists() else d.RECORD.read_bytes()
    record=json.loads(original)
    assert record['status']=='HOLD' and record['error']=='Canonical plan did not record all steps passed'
    assert record['canonical_cli']['exit_code']==0 and record['lifecycle_launched'] is True
    d.record=record
    assert not (d.OUT/'core-boundary.json').exists(), 'Successful boundary already done; do not repeat'
    assert not (d.OUT/'core-role-probe-distribution.mjs.json').exists(), 'Terminal role probe already done; do not repeat'
    if not history.exists():
        history.write_bytes(original)
    diagnostic={'status':'root_identified','recorded_at':datetime.now(UTC).isoformat(),
                'original_error_record':{'path':str(history),'sha256':d.sha(original)},
                'root':'Task harness mistook static plan.steps for execution records. Static entries intentionally have no status; actual committed checkpoint/result files carry status.',
                'product_failure':False,'canonical_run_repeated':False,'recovery':'Read actual checkpoint/result for each static step and verify command evidence filenames against canonical command tuple SHA256. Continue smoke only after every actual result passed.'}
    d.write(d.OUT/'collector-diagnostic.json',diagnostic)
    plan_path=d.ART/'plan.json';plan=json.loads(plan_path.read_bytes())
    rows=[]
    expected={
        'animetta-pull':('docker','compose','pull','--include-deps','animetta'),
        'animetta-image-status':('docker','image','inspect',d.IMAGE,'--format','{{json .RepoDigests}}'),
        'animetta-start':('docker','compose','up','-d','--no-build','animetta'),
    }
    # Definition metadata only: canonical preflight constructs its command; never execute it again.
    rspec=importlib.util.spec_from_file_location('canonical_readonly_metadata',d.ROOT/'scripts/runtime_lifecycle.py')
    runtime=importlib.util.module_from_spec(rspec)
    import sys
    sys.modules[rspec.name]=runtime
    rspec.loader.exec_module(runtime)
    expected['host-tts-preflight']=tuple(runtime._preflight(wait=False))
    assert all(not any(str(x).lower() in {'node','node.exe','pnpm','pnpm.exe','build'} for x in cmd) for cmd in expected.values())
    for step in plan['steps']:
        sid=step['id'];cp_path=d.ART/'steps'/sid/'checkpoint.json'
        cp=json.loads(cp_path.read_bytes());result_path=Path(cp['result_reference'])
        result=json.loads(result_path.read_bytes())
        row={'id':sid,'action_kind':step['action_kind'],'status':result['status'],
             'started_at':result['started_at'],'feedback_at':result['feedback_at'],
             'elapsed_seconds':result['elapsed_seconds'],'progress_summary':result['progress_summary'],
             'checkpoint':{'path':str(cp_path),'sha256':d.sha(cp_path.read_bytes())},
             'actual_result':{'path':str(result_path),'sha256':d.sha(result_path.read_bytes())},
             'evidence':[]}
        if sid in expected:
            argv=expected[sid]
            expected_name='command-'+d.sha(repr(argv).encode())[:12]+'.log'
            assert any(Path(p).name==expected_name for p in result['evidence_refs']), 'Command receipt mismatch: '+sid
            row['actual_command']=list(argv)
            row['command_binding']='Canonical run_command log filename hashes repr(actual command tuple); matches this exact expected argv'
        elif sid=='default-log-check':
            row['command_template']=['docker','compose','logs','--since','<per-window UTC timestamp>','--tail','200','animetta']
            row['command_binding']='Actual passed log-check result and immutable canonical check_logs source; dynamic timestamp is not independently recorded as an argv receipt'
        for p in result['evidence_refs']:
            ref=Path(p)
            row['evidence'].append({'path':str(ref),'bytes':ref.stat().st_size,'sha256':d.sha(ref.read_bytes())})
        rows.append(row)
    proof={'run_id':d.RUN,'status':'passed' if all(s['status']=='passed' for s in rows) else 'HOLD',
           'canonical_exit_code':0,'static_plan':{'path':str(plan_path),'sha256':d.sha(plan_path.read_bytes())},
           'actual_steps':rows,'all_actual_steps_passed':all(s['status']=='passed' for s in rows),
           'actual_pull_inspect_up_no_build':[s['actual_command'] for s in rows if s['id'] in ('animetta-pull','animetta-image-status','animetta-start')],
           'controller_node_pnpm_build_commands':[],'RVC_steps':[],
           'metadata_mapping_source':{'path':str(d.ROOT/'scripts/runtime_lifecycle.py'),'sha256':d.sha((d.ROOT/'scripts/runtime_lifecycle.py').read_bytes())},
           'browser_Node_harness_runs_only_after_controller_terminal':True}
    extra=d.OUT/'collector-command-binding-HOLD.json'
    if extra.exists():
        diagnostic['secondary_parser_failure']={'path':str(extra),'sha256':d.sha(extra.read_bytes()),'root':'Canonical check_logs dynamically adds --since timestamp and --tail200. Exact pull/inspect/up commands remain hash bound; log check records template and actual result.'}
        d.write(d.OUT/'collector-diagnostic-corrected.json',diagnostic)
    d.write(d.OUT/'canonical-receipts-corrected.json',proof)
    assert len(rows)==9 and proof['all_actual_steps_passed'], 'Actual canonical receipt failed; HOLD'
    assert not any('build' in s['action_kind'] or 'rvc' in s['id'] for s in rows)
    d.record=record
    d.record['historical_collector_failure']=diagnostic
    d.record.pop('error',None);d.record.pop('error_type',None)
    d.record['completed'].append('canonical actual 9 checkpoint/results passed; pull/inspect/up --no-build command evidence matched')
    d.checkpoint('receipt_parser_corrected_same_run',status='deployed',active_session=None,next_action='Read-only boundary/permissions then once-only role; no lifecycle')


def main():
    try:
        receipts()
        d.r.preserve_auth()
        prep=json.loads((d.OUT/'preparation.json').read_bytes())
        d.r.ownership()
        d.boundary(prep)
        d.permissions()
        d.role()
        d.write(d.OUT/'final-identity.json',{'status':'passed','host_ownership':d.r.ownership(),
                'containers':d.containers()[1],'retained_rollback_images':d.retained_images(prep),
                'credentials_continuity':d.r.preserve_auth()})
        d.record['completed'].append('final unchanged Qwen identity/PIDs/voice/private auth and retained old images')
        d.checkpoint('complete',status='acceptance_passed',canonical_pid=None,active_probe_pid=None,active_session=None,
                     next_action='Freeze small whitelist evidence/index/secret scan for parent archive; no cleanup or Git')
    except Exception as exc:
        d.checkpoint('HOLD_original_postdeploy_failure',status='HOLD',error_type=type(exc).__name__,error=str(exc),
                     next_action='Diagnose original postdeploy failure; no repeat lifecycle or successful probe')
        raise


if __name__=='__main__':
    main()
