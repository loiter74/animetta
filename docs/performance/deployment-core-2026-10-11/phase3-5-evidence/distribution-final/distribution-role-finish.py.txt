"""Resume remaining role once; validate already collected permission results without replay."""
import ast
import importlib.util
import json
from pathlib import Path

spec=importlib.util.spec_from_file_location('distribution',Path(__file__).with_name('distribution-driver.py'))
d=importlib.util.module_from_spec(spec)
spec.loader.exec_module(d)
d.record=json.loads(d.RECORD.read_bytes())


def main():
    assert d.record['status']=='HOLD'
    assert d.record['canonical_cli']['exit_code']==0
    assert json.loads((d.OUT/'canonical-receipts-corrected.json').read_bytes())['all_actual_steps_passed']
    assert json.loads((d.OUT/'core-boundary.json').read_bytes())['status']=='passed'
    assert not (d.OUT/'core-role-probe-distribution.mjs.json').exists(), 'Never repeat terminal role probe'
    p=json.loads((d.OUT/'formal-permissions.json').read_bytes())
    assert len(p['requests'])==20
    for x in p['requests']:
        expected=200 if x['authenticated_request'] or x['url'].endswith('/health') else 401
        body_status={'/health':'ok','/ready':'ready'}.get('/'+x['url'].rsplit('/',1)[-1])
        # Original loop checked numeric HTTP status before body status shadowed the field.
        if isinstance(x['status'],str):
            assert expected==200 and x['status']==body_status
            x['body_status']=x.pop('status')
            x['HTTP_status_evidence']='Original driver per-request numeric status assertion passed; numeric field not preserved because body status overwrote it'
        else:
            assert x['status']==expected
        assert 'text/html' not in x['content_type']
    assert p['invalid_origin']['status']==400
    sock=json.loads((d.OUT/'permissions-distribution.mjs.json').read_bytes())
    assert sock['status']=='passed' and sock['invalid_token_connected'] is False
    code=ast.literal_eval(sock['rejection'])['code']
    assert code=='UNAUTHORIZED', 'Actual semantic auth rejection mismatch'
    p.update(status='passed',invalid_socket_token=sock,actual_rejection_code=code,
             recovery='No HTTP or Socket request repeated. Decode actual Socket.IO error message dictionary, preserving strict UNAUTHORIZED code and connection=false.')
    d.write(d.OUT/'formal-permissions-corrected.json',p)
    original=d.OUT/'permissions-HOLD-original.json'
    d.write(d.OUT/'permissions-collector-diagnostic.json',{'root':'Harness compared raw Socket.IO message to bare string; actual message serializes the structured {code: UNAUTHORIZED} dictionary.',
            'original_HOLD_path':str(original),'original_HOLD_sha256':d.sha(original.read_bytes()),
            'semantic_permission_failure':False,'probe_repeated':False,'actual_rejection_code':code,
            'field_comparison':{'raw_message_expected':'UNAUTHORIZED','actual_raw_message':sock['rejection'],'actual_parsed_code':code,'invalid_token_connected':sock['invalid_token_connected']},
            'secondary_resume_assertion':{'expected':'numeric status for every saved entry','actual':'health body status ok / ready body status ready shadowed saved numeric HTTP status',
             'root':'HTTP and JSON body both use status; original request guard checked numeric status successfully before serialization shadow.',
             'limit':'Original raw kept unchanged. Corrected health/ready entries expose body_status; missing numeric field is explicitly labeled, not synthesized. No successful request/probe rerun.'}})
    d.record['historical_permission_collector_failure']={'path':str(original),'sha256':d.sha(original.read_bytes())}
    d.record.pop('error',None);d.record.pop('error_type',None)
    d.record['completed'].append('20 formal/direct API checks, actual invalid Origin400 and structured Socket UNAUTHORIZED/connection=false verified from original raw')
    d.checkpoint('permissions_existing_raw_verified',status='deployed',next_action='One remaining role/native/ordinary continuity probe; no lifecycle or permission replay')
    d.r.preserve_auth();d.r.ownership()
    try:
        d.role()
        prep=json.loads((d.OUT/'preparation.json').read_bytes())
        d.write(d.OUT/'final-identity.json',{'status':'passed','host_ownership':d.r.ownership(),
                'containers':d.containers()[1],'retained_rollback_images':d.retained_images(prep),
                'credentials_continuity':d.r.preserve_auth()})
        d.record['completed'].append('final unchanged Qwen identity/PIDs/voice/private auth and retained old images')
        d.checkpoint('complete',status='acceptance_passed',canonical_pid=None,active_probe_pid=None,active_session=None,
                     next_action='Freeze small whitelist evidence/index/secret scan for parent archive; no cleanup or Git')
    except Exception as exc:
        d.checkpoint('HOLD_role_original_failure',status='HOLD',error_type=type(exc).__name__,error=str(exc),
                     next_action='Read original role diagnostics; never repeat terminal role or lifecycle')
        raise


if __name__=='__main__':
    main()
