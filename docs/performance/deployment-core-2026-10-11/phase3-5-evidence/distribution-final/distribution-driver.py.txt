def sha(b):
    return hashlib.sha256(b).hexdigest()


def sanitize(b):
    for s in SECRETS:
        if len(s) >= 8:
            b = b.replace(s.encode(), b'<redacted>')
    return b


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(sanitize((json.dumps(value, ensure_ascii=False, indent=2)+'\n').encode()))


def checkpoint(step, **kw):
    record.update(current_step=step, updated_at=datetime.now(UTC).isoformat(), **kw)
    write(RECORD, record)
    # Re-read and merge only this owner's namespace; preserve parent's concurrent fields.
    for _ in range(3):
        original = STATE.read_bytes()
        d = json.loads(original)
        section = d.setdefault('distribution_final', {})
        section.update(status=record['status'], current_step=step, run_id=RUN,
                       image_digest=IMAGE, expected_release_revision=REV,
                       publication_status='CI38126403988 completed/success; published digest verified by parent',
                       updated_at=record['updated_at'], record_path=str(RECORD), evidence_root=str(OUT),
                       completed=list(record['completed']), next_action=record.get('next_action'),
                       canonical_pid=record.get('canonical_pid'), active_probe_pid=record.get('active_probe_pid'),
                       driver_pid=os.getpid(), active_session=record.get('active_session'),
                       lifecycle_or_pull_executed=record.get('lifecycle_launched', False),
                       run_id_status='launched_once' if record.get('lifecycle_launched') else 'reserved_not_launched')
        raw = sanitize((json.dumps(d, ensure_ascii=False, indent=2)+'\n').encode())
        if b'\r\n' in original:
            raw = raw.replace(b'\n', b'\r\n')
        if STATE.read_bytes() == original:
            STATE.write_bytes(raw)
            break
    else:
        raise RuntimeError('Concurrent state update: do not overwrite parent')
    print(json.dumps({'run_id': RUN, 'status': record['status'], 'step': step,
                      'canonical_pid': record.get('canonical_pid'), 'probe_pid': record.get('active_probe_pid')}, ensure_ascii=False), flush=True)


def command(argv, timeout=30):
    c = subprocess.run(argv, cwd=ROOT, env=ENV, capture_output=True, timeout=timeout)
    if c.returncode:
        raise RuntimeError('Read-only command failed: '+str(argv[:3])+' '+sanitize(c.stderr).decode('utf-8','replace')[:500])
    return c.stdout.decode('utf-8', 'replace')


def http(url, authenticated=True, key='ANIMETTA_ACCESS_TOKEN', headers=None):
    h = {'Authorization': 'Bearer '+ENV[key]} if authenticated else {}
    h.update(headers or {})
    start = time.perf_counter()
    try:
        response = urlopen(Request(url, headers=h), timeout=15)
    except HTTPError as e:
        response = e
    with response:
        raw = response.read()
        ct = response.headers.get('Content-Type', '')
        return {'url': url, 'status': response.status, 'bytes': len(raw), 'sha256': sha(raw),
                'elapsed_seconds': time.perf_counter()-start, 'content_type': ct,
                'body': json.loads(raw) if 'json' in ct else None,
                'assets': re.findall(r'(?:src|href)=[\"\']([^\"\']+\.(?:js|css)[^\"\']*)', raw.decode('utf-8','replace'))}

def node_probe(name, timeout):
    script=OUT/'probes'/name
    child=subprocess.Popen(['node',str(script)],cwd=script.parent,env=ENV,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    checkpoint('actual_browser_probe' if name.startswith('core-role') else 'socket_permission_probe',active_probe_pid=child.pid)
    try:
        out,err=child.communicate(timeout=timeout)
    except subprocess.TimeoutExpired:
        # Only this owned evidence helper is ended; never touch product or host services.
        child.kill();out,err=child.communicate()
        (OUT/(name+'.partial.log')).write_bytes(sanitize(out+err))
        raise RuntimeError('Owned browser helper exceeded bounded timeout; original partial output preserved')
    out,err=sanitize(out),sanitize(err)
    (OUT/(name+'.stderr.log')).write_bytes(err)
    (OUT/(name+'.json')).write_bytes(out)
    checkpoint('browser_probe_terminal',active_probe_pid=None)
    assert child.returncode==0, 'Probe process failed; preserve terminal result'
    return json.loads(out)
