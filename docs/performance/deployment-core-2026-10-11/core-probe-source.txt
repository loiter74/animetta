import asyncio
import importlib.metadata
import json
import os
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, str(Path.cwd() / "src"))
os.environ.update(ANIMETTA_HOST="127.0.0.1", ANIMETTA_PORT="12394")
for name in ("MIMO_API_KEY", "DASHSCOPE_API_KEY"):
    os.environ.pop(name, None)

import animetta.orchestration.server.websocket
from animetta.config.manifest import RuntimeCapabilities, load_effective_config
from animetta.runtime.session_context import ServiceContext
from animetta.memory.v2.system import LivingMemorySystem

async def probe():
    config = load_effective_config(profile="test")
    config = config.model_copy(update={"runtime": config.runtime.model_copy(update={"capabilities": RuntimeCapabilities()})})
    ctx = ServiceContext()
    await ctx.load_from_config(config)
    assert ctx.llm_engine is not None and ctx.tts_engine is not None
    assert ctx.asr_engine is None and ctx.vad_engine is None and ctx.audio_processor is None
    with tempfile.TemporaryDirectory(prefix="anima-core-probe-") as temporary:
        memory = LivingMemorySystem(db_path=str(Path(temporary) / "memory.sqlite"))
        await memory.initialize()
        await memory.shutdown()
    await ctx.close()
    names = {d.metadata["Name"].lower() for d in importlib.metadata.distributions()}
    prohibited = {"torch", "transformers", "peft", "bilibili-api-python", "langchain-community", "chromadb"}
    assert not names & prohibited, names & prohibited
    modules = ("animetta.services.singing.svc_pipeline", "animetta.services.vad.silero_vad", "animetta.services.vad.mimo_vad", "animetta.orchestration.server.handlers.earth_handlers")
    assert not set(modules) & sys.modules.keys()
    print(json.dumps({"python": sys.version, "packages": len(names), "core_import": True, "core_context": True, "memory_v2": True, "optional_modules_loaded": False, "providers": list(config.runtime.required_services), "real_provider_playback": "separate runtime acceptance required"}))

asyncio.run(probe())
