"""Bounded read-only 503->200 observer; it never starts/stops a service."""
import json
import subprocess
import time
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen

from pathlib import Path
import importlib.util
import argparse

O = Path(__file__).parent
spec = importlib.util.spec_from_file_location('after_stages', O / 'run-after.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--run-id', choices=['llmr-fault01', 'llmr-fault02'], default='llmr-fault01')
    run_id = parser.parse_args().run_id
    prefix = 'fault02' if run_id == 'llmr-fault02' else 'fault'
    # Parent driver supplies the privately inherited environment before creating the app.
    g = m.helpers(current_credentials=False)
    path = O / (prefix + '-observation.json')
    assert not path.exists(), 'Original observer exists; inspect it, never repeat'
    rows = []
    first = None
    backend = None
    deadline = time.monotonic() + 300
    normal_image = g['record']['warmup']['containers'][0]['image']
    shim_sha = g['sha']((O / 'fault/sitecustomize.py').read_bytes())
    assert shim_sha == json.loads((O / 'acceptance-plan.json').read_bytes())['files']['fault/sitecustomize.py'], 'Shim changed'
    result = {'status': 'observing', 'run_id': run_id, 'observations': rows, 'container': None}
    try:
        while time.monotonic() < deadline:
            c = subprocess.run(['docker', 'inspect', 'anima-animetta-1'], capture_output=True, timeout=5)
            if c.returncode == 0:
                current = json.loads(c.stdout)[0]
                binds = [x for x in current['Mounts'] if x['Destination'] == '/app/src/sitecustomize.py']
                if current['Config']['Image'] == m.NORMAL and binds and current['State']['Running']:
                    assert len(binds) == 1 and binds[0]['Type'] == 'bind' and binds[0]['RW'] is False, 'Fault bind must be read-only'
                    assert binds[0]['Source'].replace('\\', '/').lower().endswith('/llm-startup-recovery-20261011/fault/sitecustomize.py'), 'Unexpected shim source'
                    assert current['Image'] == normal_image, 'Normal image changed'
                    identity = {k: current[k] for k in ['Id', 'Image', 'RestartCount']}
                    identity['StartedAt'] = current['State']['StartedAt']
                    if first is None:
                        first = identity
                        deadline = min(deadline, time.monotonic() + 90)
                        result['container'] = identity
                        result.update(shim_source_SHA256=shim_sha, bind=binds[0], build_fingerprint=current['Config']['Labels'].get('org.animetta.build-fingerprint'),
                                      fault_runtime_fingerprint=current['Config']['Labels']['org.animetta.runtime-fingerprint'])
                        assert result['fault_runtime_fingerprint'] != 'untracked', 'Missing canonical fault runtime fingerprint'
                        assert result['build_fingerprint'] == g['record']['normal_identity']['build_fingerprint'], 'Normal build fingerprint changed'
                        result['normal_runtime_fingerprint'] = g['record']['normal_identity']['runtime_fingerprint']
                        assert result['fault_runtime_fingerprint'] != result['normal_runtime_fingerprint'], 'Fault runtime fingerprint did not change'
                    assert first == identity and current['RestartCount'] == 0, 'Container restarted/replaced'
                    transport_error = None
                    try:
                        with urlopen(Request('http://localhost/ready', headers={'Authorization': 'Bearer ' + g['ENV']['ANIMETTA_ACCESS_TOKEN']}), timeout=1) as r:
                            status, body = r.status, json.loads(r.read())
                    except HTTPError as ex:
                        status = ex.code
                        data = ex.read()
                        try:
                            body = json.loads(data)
                        except json.JSONDecodeError:
                            body = {}
                    except (URLError, TimeoutError, ConnectionError) as ex:
                        status, body = None, {}
                        transport_error = type(ex).__name__
                    row = {'observed_at': g['now'](), 'epoch_seconds': time.time(), 'status': status, 'ready': body.get('ready'),
                           'transport_error': transport_error,
                           'components': {k: {n: v.get(n) for n in ['state', 'ready', 'reason', 'required']} for k, v in body.get('components', {}).items()}}
                    rows.append(row)
                    if status in [503, 200] and (backend is None or status == 200):
                        code = "import pathlib,json; print(json.dumps([int(p.name) for p in pathlib.Path('/proc').iterdir() if p.name.isdigit() and b'animetta.core.socketio_server' in (p/'cmdline').read_bytes() and int(p.name)!=__import__('os').getpid()]))"
                        pids = json.loads(g['cmd'](['docker', 'exec', current['Id'], 'python', '-c', code]))
                        assert len(pids) == 1, 'Backend PID ambiguous'
                        if backend is None:
                            backend = pids[0]
                        assert backend == pids[0], 'Backend replaced'
                    g['write'](path, result)
                    if status == 200:
                        assert body.get('ready') is True
                        assert any(x['status'] == 503 and x['components'].get('llm', {}).get('ready') is not True for x in rows), 'No strict LLM nonready observation'
                        logs = subprocess.run(['docker', 'logs', '--timestamps', current['Id']], capture_output=True, timeout=15)
                        data = g['clean'](logs.stdout + logs.stderr)
                        (O / (prefix + '-startup.log')).write_bytes(data)
                        events = [json.loads(s.split('LLMR_FAULT ', 1)[1]) for s in data.decode('utf8', 'replace').splitlines() if 'LLMR_FAULT ' in s]
                        cancelled = [x for x in events if x['event'] == 'first_delay_cancelled']
                        assert len(cancelled) == 1 and cancelled[0]['pid'] == backend, 'First real deadline did not cancel injected delay'
                        assert not any(x['event'] == 'first_delay_uncancelled' for x in events)
                        assert any(x['event'] == 'native_send_complete' and x['status_code'] == 200 and x['pid'] == backend for x in events), 'No real native transport success'
                        recovered = row['epoch_seconds'] - cancelled[0]['epoch_seconds']
                        assert 0 <= recovered <= 60, 'Recovery exceeded frozen 60-second prediction'
                        result.update(status='passed', backend_pid=backend, shim_events=events, timeout_to_ready_seconds=recovered)
                        g['write'](path, result)
                        return
            time.sleep(.2)
        raise TimeoutError('Bounded observer did not see same-process strict readiness recovery')
    except Exception as ex:
        result.update(status='HOLD', error_type=type(ex).__name__, error=str(ex))
        g['write'](path, result)
        raise


if __name__ == '__main__':
    main()
