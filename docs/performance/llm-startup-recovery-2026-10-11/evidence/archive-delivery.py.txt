"""Archive completed measurements without changing their captured bytes."""

import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
SOURCE = Path(__file__).parent
DESTINATION = ROOT / "docs/performance/llm-startup-recovery-2026-10-11"


def main() -> None:
    state = json.loads((SOURCE / "state.json").read_bytes())
    assert not any(state.get(key) for key in (
        "active_pid", "driver_pid", "active_tool_session", "observer_pid"
    )), "Evidence still has an active writer"
    stages = json.loads((SOURCE / "after.json").read_bytes())["stages"]
    assert all(stages[key]["status"] == "passed" for key in (
        "freeze", "warmup", "after01", "after02", "fault", "role", "restore"
    )), "Required runtime evidence is incomplete"
    destination = DESTINATION / "evidence"
    destination.mkdir(parents=True, exist_ok=True)
    entries = []
    for path in sorted(SOURCE.rglob("*")):
        if not path.is_file() or "__pycache__" in path.parts or path == Path(__file__):
            continue
        relative = path.relative_to(SOURCE)
        if path.suffix not in {".json", ".log", ".txt", ".py", ".mjs", ".yml", ".png", ".Dockerfile"}:
            continue
        archived = relative
        if path.suffix in {".py", ".mjs", ".Dockerfile"}:
            archived = relative.with_name(relative.name + ".txt")
        data = path.read_bytes()
        target = destination / archived
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
        entries.append({
            "source": relative.as_posix(), "archive": archived.as_posix(),
            "bytes": len(data), "SHA256": hashlib.sha256(data).hexdigest(),
        })
    index = {"schema_version": 1, "files": entries, "file_count": len(entries),
             "total_bytes": sum(item["bytes"] for item in entries),
             "policy": "Captured bytes are preserved; executable helper carriers use .txt."}
    (destination / "index.json").write_text(json.dumps(index, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    (DESTINATION / ".gitattributes").write_text(
        "* -text whitespace=trailing-space,space-before-tab,cr-at-eol\n"
        "# Preserve native captured Redis logs, including trailing spaces.\n"
        "*.log whitespace=-trailing-space,space-before-tab,cr-at-eol\n", encoding="utf-8")
    print(json.dumps({"file_count": index["file_count"], "total_bytes": index["total_bytes"]}))


if __name__ == "__main__":
    main()
