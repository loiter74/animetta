containers = json.loads(g['cmd'](['docker', 'inspect', 'anima-animetta-1', 'anima-redis-1']))

ready = {}
for base in ['http://localhost', 'http://127.0.0.1:12394']:
    with urlopen(Request(base + '/ready', headers={'Authorization': 'Bearer ' + g['ENV']['ANIMETTA_ACCESS_TOKEN']}), timeout=10) as response:
        b = response.read()
        body = json.loads(b)

qwen = []
for x in json.loads((O / 'current-inventory.json').read_bytes())['Qwen_processes']:
    p = psutil.Process(x['pid'])
    assert p.create_time() == x['creation_time'] and p.cmdline() == x['command_line']
    qwen.append({'pid': p.pid, 'creation_time': p.create_time(), 'command_line': p.cmdline()})
active = []
for p in psutil.process_iter(['name', 'cmdline']):
    try:
        args = p.info['cmdline'] or []
        if p.pid == os.getpid() or not args:
            continue
        name = (p.info['name'] or '').lower()
        entries = [x for x in args[1:] if x.endswith(('.py', '.mjs'))]
        if (name.startswith('python') or name == 'node.exe') and any(
            pathlib.Path(x).name == 'runtime_lifecycle.py' or
            ('llm-startup-recovery-20261011' in x and pathlib.Path(x).name in {
                'run-baseline.py', 'run-after.py', 'observe-fault.py', 'run-role-recovered.py',
                'core-role-probe-recovered.mjs', 'core-role-probe-fault02.mjs'}) for x in entries
        ):
            active.append({'pid': p.pid, 'command_line': args})
    except (psutil.NoSuchProcess, psutil.AccessDenied):
        pass
assert not active, 'Actual task interpreter/script still active'
final = {'status': 'passed', 'collected_at': g['now'](), 'normal_image': m.NORMAL,

# Close existing canonical evidence references only; do not copy static/feedback plans.
receipts = []
for record_name in ['baseline.json', 'after.json']:
    doc = json.loads((O / record_name).read_bytes())
    for run in doc['runs']:
        for step in run['steps']:
            for ref in step.get('evidence_refs') or []:
                src = pathlib.Path(ref)
                if not src.is_file():
                    continue
                b = src.read_bytes()
                assert len(b) < 2_000_000, 'Oversized artifact excluded; inspect closure need before copying'
                safe = g['clean'](b)
                dest = O / ('baseline' if record_name == 'baseline.json' else 'after') / 'evidence' / run['run_id'] / (step['step_id'] + '-' + src.name)
                dest.parent.mkdir(parents=True, exist_ok=True)
                if dest.exists():
                    assert dest.read_bytes() == safe
                else:
                    dest.write_bytes(safe)
                receipts.append({'source_path': str(src), 'source_SHA256': g['sha'](b),
                                 'path': str(dest.relative_to(O)).replace('\\', '/'),
                                 'SHA256': g['sha'](safe), 'redacted': safe != b})
g['write'](O / 'canonical-evidence-closure.json', {'files': receipts})
